cmake_minimum_required(VERSION 3.28)
project(paleo_native_fusion LANGUAGES CXX)
# Compile the exact production client and its QtTest without the GIS/UI closure.
get_filename_component(PALEO_SOURCE_ROOT "${CMAKE_CURRENT_LIST_DIR}/../.." ABSOLUTE)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_SCAN_FOR_MODULES OFF)
set(CMAKE_AUTOMOC ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
find_package(Qt6 REQUIRED COMPONENTS Core Network Test)
find_program(PALEO_PYTHON3 NAMES python3 python REQUIRED)
if(MSVC)
  add_compile_options(/utf-8 /W4)
  add_compile_definitions(NOMINMAX)
else()
  add_compile_options(-Wall -Wextra -Werror=narrowing)
endif()
enable_testing()
function(add_paleo_test name)
  cmake_parse_arguments(PARSE_ARGV 1 apt "" "" "LIBS")
  add_executable(${name} "${PALEO_SOURCE_ROOT}/tests/${name}.cpp")
  target_link_libraries(${name} PRIVATE ${apt_LIBS} Qt6::Test)
  add_test(NAME ${name} COMMAND ${name} -o "${CMAKE_CURRENT_BINARY_DIR}/${name}.txt,txt")
  set_tests_properties(${name} PROPERTIES TIMEOUT 60)
endfunction()
include("${PALEO_SOURCE_ROOT}/cmake/extra-seismic-fusion.cmake")
